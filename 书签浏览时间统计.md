>书签中存放待学习内容

```dataviewjs
// 1. 获取 Obsidian 原生书签插件的数据
const bookmarkPlugin = app.internalPlugins?.getPluginById?.("bookmarks");

if (!bookmarkPlugin || !bookmarkPlugin.enabled) {
    dv.paragraph("⚠️ 未检测到开启的核心插件「书签 (Bookmarks)」，请先在设置中开启该插件。");
} else {
    const items = bookmarkPlugin.instance?.items || [];

    // 2. 把一组书签笔记转换成可渲染的行（读取浏览时间并倒序排序）
    function buildRows(fileItems) {
        return fileItems.map(item => {
            const page = dv.page(item.path);
            if (!page) return null;

            // 优先读取属性中的 viewed，其次 last_read
            const lastRead = page.viewed || page.last_read || "未记录";

            return {
                link: page.file.link,
                lastRead,
                sortKey: page.viewed || page.last_read || ""
            };
        }).filter(Boolean).sort((a, b) => {
            if (a.sortKey > b.sortKey) return -1;
            if (a.sortKey < b.sortKey) return 1;
            return 0;
        });
    }

    // 3. 递归按书签文件夹（分组）罗列笔记
    function render(list, depth) {
        let pending = []; // 当前层级下直属的笔记，遇到子分组前先输出

        const flush = () => {
            if (pending.length === 0) return;
            // 根目录下直属的笔记（不属于任何分组）单独标一个标题
            if (depth === 0) dv.header(2, "📁 未分组");
            const rows = buildRows(pending);
            if (rows.length) {
                dv.table(
                    ["书签笔记", "上次浏览时间"],
                    rows.map(r => [r.link, r.lastRead])
                );
            }
            pending = [];
        };

        for (const item of list) {
            if (item.type === "file") {
                pending.push(item);
            } else if (item.type === "group") {
                flush(); // 先输出本层已收集的笔记
                const title = item.title || item.name || "未命名分组";
                dv.header(Math.min(6, depth + 2), `📁 ${title}`);
                render(item.items || [], depth + 1);
            }
        }
        flush();
    }

    // 4. 统计是否真的存在书签笔记
    function countFiles(list) {
        let n = 0;
        for (const item of list) {
            if (item.type === "file") n++;
            else if (item.type === "group") n += countFiles(item.items || []);
        }
        return n;
    }

    if (countFiles(items) === 0) {
        dv.paragraph("ℹ️ 当前书签栏中还没有收藏任何笔记。");
    } else {
        render(items, 0);
    }
}
```
