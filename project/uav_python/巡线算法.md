def line(self, image, cont_cent):
	ls = self.cfg.line_slicing
	lf = self.cfg.line_follow
	center_x = ls.center_x
	num_slices = ls.num_slices
	sl = int(image.shape[0] / num_slices)

	# 计算每个块中心点间的偏移角度
	line_angle1 = math.atan2(
		cont_cent[1][0] - cont_cent[0][0],
		cont_cent[1][1] + sl - cont_cent[0][1],
	)
	line_angle1 = int(180 * line_angle1 / math.pi)

	line_angle2 = math.atan2(
		cont_cent[2][0] - cont_cent[1][0],
		cont_cent[2][1] + sl * 2 - cont_cent[1][1] + sl,
	)
	line_angle2 = int(180 * line_angle2 / math.pi)

	line_angle3 = math.atan2(
		cont_cent[3][0] - cont_cent[2][0],
		cont_cent[3][1] + sl * 3 - cont_cent[2][1] + sl * 2,
	)
	line_angle3 = int(180 * line_angle3 / math.pi)
	self.is_straight = (
		max(abs(line_angle1), abs(line_angle2), abs(line_angle3)) < 20
	)
	offset_center = 0
	cnt = 0
	speed_row = 0
	# 计算各切片中心点的平均偏移量，作为整体的偏移量输入pid进行调整
	for i in range(num_slices):
		if cont_cent[i][0] != 0:
			offset_center += cont_cent[i][0] - center_x
			cnt += 1
	logger.info(f"Count: {cnt}")
	if self.show_video:
		cv2.putText(image, f"Count: {cnt}", (10, 30), cv2.FONT_HERSHEY_SIMPLEX, 1, 0, 2)
	if cnt != 0:
		offset_center = offset_center // cnt
		if cnt == num_slices:
			self.is_begin = 0
			self.offset_o = offset_center
	self.offset_center_d = offset_center
