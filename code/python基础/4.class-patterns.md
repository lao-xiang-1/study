# Python 高级语法（四）：类与对象高级模式

## 1. 行为重载：让对象支持运算符和调用

### 1.1 容器方法

```python
# pcx/core/_static.py:92-110 —— StaticParam 重载容器方法
class StaticParam(BaseParam):
    def __contains__(self, __key: str) -> bool:
        """支持 'key in static_param' 操作"""
        return __key in self._static_value

    def __iter__(self) -> Any:
        """支持 'for x in static_param' 操作"""
        return iter(self._static_value)

    def __len__(self) -> int:
        """支持 len(static_param)"""
        return len(self._static_value)

    def __getitem__(self, __idx: Any) -> Any:
        """支持 static_param[idx]"""
        return self._static_value.__getitem__(__idx)

    def __setitem__(self, __idx: Any, value: Any) -> Any:
        """支持 static_param[idx] = value"""
        return self._static_value.__setitem__(__idx, value)
```

```python
# pcx/core/_parameter.py:301-312 —— ParamDict 的容器方法
class ParamDict(DynamicParam):
    def __getitem__(self, __key: str) -> Any:
        return self._value[__key]

    def __setitem__(self, __key: str, __value: jax.Array) -> None:
        if self._value is None:
            self._value = {}
        self._value[__key] = __value

    def __contains__(self, __key: str) -> bool:
        return __key in self._value
```

### 1.2 数值运算符重载

```python
# pcx/core/_parameter.py:133-267 —— Param 重载所有算术运算符
class Param(DynamicParam):
    def __add__(self, __other):       # a + b
        return self._value.__add__(get(__other))

    def __radd__(self, __other):      # b + a（当 a 是 Param 时）
        return self._value.__radd__(get(__other))

    def __iadd__(self, __other):      # a += b
        self._value = self._value.__add__(get(__other))
        return self

    def __mul__(self, __other):       # a * b
        return self._value.__mul__(get(__other))

    def __matmul__(self, __other):    # a @ b（矩阵乘法）
        return self._value.__matmul__(get(__other))

    # ... 还重载了 sub, div, mod, pow, and, or, xor, lshift, rshift 等
```

**效果**：使 `Param` 对象可以直接参与数学运算，如同它们是裸数组一样：
```python
w = Param(jnp.array([1.0, 2.0]))
result = w * 2 + 1  # 直接运算，无需 w.get()
```

class M:
    def __invert__(self) -> "_M_not":    # ~M(...) —— 逻辑非

### 1.4 `__call__` —— 可调用对象

定义 `__call__` 后，实例可以像函数一样被调用。

```python
# pcx/predictive_coding/_vode.py:204-236
class Vode(EnergyModule):
    def __call__(
        self, u: jax.Array | None, rkg: RandomKeyGenerator = RKG, output="h", **kwargs
    ) -> jax.Array | Any:
        """使 Vode 实例可调用：vode(activation)"""
        if u is not None:
            self.set("u", u, rkg)

        for _k, _v in kwargs.items():
            self.set(_k, _v, rkg)

        if (h := self.get("u")) is not None:
            self.shape.set(h.shape)

        if output is None:
            return self
        else:
            return self.get(output, rkg=rkg)
```

```python
# pcx/core/_static.py:112-114 —— StaticParam 也是可调用的
class StaticParam(BaseParam):
    def __call__(self, *args, **kwds) -> Any:
        """如果是函数，直接调用"""
        return self._static_value(*args, **kwds)

# 用法：
m = Module()
m.energy_fn = static(some_function)
m.energy_fn(x, y)  # 等价于 some_function(x, y)
```

### 1.5 `__new__` —— 对象创建

`__new__` 在 `__init__` 之前调用，负责**创建实例**。在元类中使用 `__new__` 而非 `__init__` 是因为前者可以控制**是否**创建实例以及**返回什么**。

```python
# pcx/core/_parameter.py:30-42 —— 元类的 __new__
class _BaseParamMeta(abc.ABCMeta):
    def __new__(mcs, name, bases, dct):
        _cls = super().__new__(mcs, name, bases, dct)  # 先创建类
        # ... 注册到 JAX ...
        return _cls  # 返回修改后的类
```

普通类中的 `__new__`（用于 deserialization）：

```python
# pcx/core/_parameter.py:59-67 —— 对象重建
@staticmethod
def unflatten_parameter(
    aux_data: Dict[str, Any], children: Any, *, cls: Type["BaseParam"]
) -> "BaseParam":
    _param = object.__new__(cls)    # 绕过 __init__，直接创建裸对象
    _param.__dict__ = dict.copy(aux_data)   # 手动恢复属性
    _param._value = children[0]
    return _param
```

**关键**：`object.__new__(cls)` 只分配内存，不调用 `__init__`。这允许完全手动控制对象的重建过程。

### 1.6 `__getattr__` —— 属性代理

`__getattr__` 在常规属性查找失败时被调用，可用于代理。

```python
# pcx/core/_static.py:88-90
class StaticParam(BaseParam):
    def __getattr__(self, __name: str) -> Any:
        """将属性访问代理到被包装的静态值"""
        return getattr(self._static_value, __name)
# 效果：static_param.any_attr 等价于 static_param._static_value.any_attr
```

```python
# pcx/core/_parameter.py:278-279
class Param(DynamicParam):
    def __getattr__(self, __name):
        """将属性访问代理到被包装的 JAX 数组，如 .shape, .dtype"""
        return getattr(self._value, __name)
# 效果：param.shape 等价于 param._value.shape
```

## 2.类的内置属性

### 2.2 `__dict__` —— 属性字典

每个实例对象都有一个 `__dict__`，存储其**动态属性**。

```python
class Person:
    species = "human"          # 类属性，不在实例 __dict__ 中
    def __init__(self, name):
        self.name = name       # 实例属性，进入 __dict__
        self._age = 0

p = Person("Alice")
print(p.__dict__)              # {'name': 'Alice', '_age': 0}
print(Person.__dict__.keys())  # 包含 'species', '__init__', ...

# 动态添加属性也会进入 __dict__
p.email = "alice@example.com"
print(p.__dict__)              # {'name': 'Alice', '_age': 0, 'email': '...'}
```

**用途**：
- 序列化时保存/恢复对象状态：`json.dumps(obj.__dict__)`
- 遍历所有实例属性：`for k, v in obj.__dict__.items()`
- 配合 `__new__` 手动重建对象：`obj.__dict__.update(saved_dict)`

### 2.3 `__slots__` —— 限制属性、节省内存

用 `__slots__` 显式声明允许的属性，**禁用 `__dict__`**，减少内存占用并加快属性访问。

```python
# pcx/core/_static.py —— 类似思想：静态值不需要动态属性
class Point:
    __slots__ = ("x", "y")     # 只允许 x 和 y 两个属性

    def __init__(self, x, y):
        self.x = x
        self.y = y

p = Point(1, 2)
p.x = 3                        # OK
# p.z = 4                      # AttributeError! 'Point' object has no attribute 'z'
# print(p.__dict__)            # AttributeError! 没有 __dict__
```

**适用场景**：
- 大量小对象（如几何点、神经网络节点）→ 显著降低内存
- 需要防止动态属性误添加 → 提前暴露拼写错误

### 2.4 `__dir__` —— 自定义 `dir()` 返回

`dir(obj)` 内部调用 `obj.__dir__()`，返回对象的属性和方法列表。可重载以控制 IDE 自动补全和交互式探索看到的内容。

```python
class Wrapper:
    def __init__(self, wrapped):
        self._wrapped = wrapped

    def __dir__(self):
        # 让 dir(wrapper) 同时显示 Wrapper 自身和被包装对象的属性
        return sorted(set(
            dir(self.__class__) +
            list(self.__dict__.keys()) +
            dir(self._wrapped)
        ))

w = Wrapper("hello")
print("zfill" in dir(w))       # True —— 来自 str 的方法
```

## 3. 元类 (Metaclass)

### 3.1 参数注册的元类

```python
# pcx/core/_parameter.py:24-42
class _BaseParamMeta(abc.ABCMeta):         # 继承自 ABCMeta，本身也是一个元类
    """
    元类：自动将所有参数类注册为 JAX pytree，实现参数的自动展平/还原。
    """

    def __new__(mcs, name, bases, dct):    # 在类被创建时调用
        _cls = super().__new__(mcs, name, bases, dct)

        # 注册到 JAX 的 pytree 系统
        jax.tree_util.register_pytree_with_keys(
            _cls,
            flatten_func=_BaseParamMeta.flatten_parameter,
            flatten_with_keys=_BaseParamMeta.flatten_parameter_with_keys,
            unflatten_func=functools.partial(
                _BaseParamMeta.unflatten_parameter, cls=_cls
            ),
        )

        return _cls

# 使用元类的类：
class BaseParam(metaclass=_BaseParamMeta):  # ← 指定元类
    ...
```

**执行时机**：当 Python 解释器读到 `class BaseParam(metaclass=_BaseParamMeta)` 时，`_BaseParamMeta.__new__` 被调用。

## 4. 抽象基类 (ABC)

`abc.ABCMeta` + `@abc.abstractmethod` 强制子类实现特定方法。
> `abc.ABCMeta`本身不强制子类实现特定方法

```python
# pcx/core/_parameter.py:70-96
class BaseParam(metaclass=_BaseParamMeta):
    """所有参数的抽象基类。"""

    def __init__(self, value: jax.Array | Any | None = None):
        self._value = value

    @abc.abstractmethod          # 子类必须实现
    def get(self):
        raise NotImplementedError()

    @abc.abstractmethod          # 子类必须实现
    def set(self, value):
        raise NotImplementedError()

    def __bool__(self):
        raise TypeError(
            "To prevent accidental errors parameters can not be used as Python bool. "
            "To check if variable is `None` use `is None` or `is not None` instead."
        )
```

> **注意**：直接实例化 `BaseParam()` 或 `_BaseTransform()` 会报错，必须通过子类实例化。
