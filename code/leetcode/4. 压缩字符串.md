```python
class Solution:
    def compress(self, chars: List[str]) -> int:
        def split_num(num: int) -> tuple(List, int):
            if num == 1:
                return [], 0
            if num < 10:
                return [str(num)], 1
            res = []
            count = 0
            while num > 0:
                cache = num % 10
                num //= 10
                res = [str(cache)] + res
                count += 1
            return res, count
        n = len(chars)
        ans = []
        char = chars[0]
        char_num = 0
        sum_count = 0
        for i in range(n):
            if chars[i] == char:
                char_num += 1
            elif chars[i] != char:
                # update ans & sum_count
                res, count = split_num(char_num)
                sum_count += count + 1
                ans.append(char)
                ans += res
                # reset char & char_num
                char = chars[i]
                char_num = 1
        # update the last char
        res, count = split_num(char_num)
        sum_count += count + 1
        ans.append(char)
        ans += res
        
        print(ans)
        print(sum_count)
        return sum_count
```
