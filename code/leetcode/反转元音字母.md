## 解法1
先遍历一遍字符串，存储元音字母。再遍历一遍来反转元音
```python
class Solution:
    def reverseVowels(self, s: str) -> str:
        vowel = []
        l = list(s)
        temp = 'aeiou'
        for index, value in enumerate(s):
            if value in temp + temp.upper():
                vowel.append((index, value))
        print(vowel)
        ans = l
        for index, (key, value) in enumerate(vowel):
            ans[key] = vowel[len(vowel) - index - 1][1]
        return ''.join(ans)
```

## 解法2
双指针，一个向前，一个向后。检测到元音则交换
```python
class Solution:
    def reverseVowels(self, s: str) -> str:
        vowel = []
        l = list(s)
        temp = 'aeiou'
        former = 0
        later = len(l) - 1
        
        while former < later:
            while l[former] not in temp + temp.upper() and former < later:
                former += 1
            while l[later] not in temp + temp.upper() and former < later:
                later -= 1
            if former >= later:
                break
            cache = l[former]
            l[former] = l[later]
            l[later] = cache
            print(former, later)
            former += 1
            later -= 1
        return ''.join(l)
```
