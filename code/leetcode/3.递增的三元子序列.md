## 暴力解法
```python
class Solution:
    def increasingTriplet(self, nums: List[int]) -> bool:
        for i, v in enumerate(nums):
            if i > 0 and i < len(nums):
                L = nums[:i]
                R = nums[i:]
                if min(L) < v and v < max(R):
                    return True
        return False
```

## 双向遍历
```python
class Solution:
    def increasingTriplet(self, nums: List[int]) -> bool:
        n = len(nums)
        if n < 3:
            return False
        
        LeftMin = [nums[0]]
        for i in nums[1:]:
            LeftMin.append(i if i < LeftMin[-1] else LeftMin[-1])
        
        RightMax = [0] * n
        RightMax[-1] = nums[-1]
        for i in range(n - 2, -1, -1):
            RightMax[i] = nums[i] if nums[i] > RightMax[i + 1] else RightMax[i + 1]

        for i in range(1, n - 1):
            if LeftMin[i - 1] < nums[i] < RightMax[i + 1]:
                return True
```

## 贪心
```python
class Solution:
    def increasingTriplet(self, nums: List[int]) -> bool:
        n = len(nums)
        first, second = nums[0], float('inf')
        for i in range(1, n):
            num = nums[i]
            if num > second:
                return True
            elif num > first:
                second = num
            else:
                first = num
        return False


```
